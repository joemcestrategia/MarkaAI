-- 1. Habilita pg_net

-- 2. FunÃ§Ã£o trigger
CREATE OR REPLACE FUNCTION sync_gcal_webhook_trigger()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
  payload jsonb;
BEGIN
  IF TG_OP = 'DELETE' THEN
    payload := jsonb_build_object(
      'type', 'DELETE',
      'table', 'appointments',
      'record', to_jsonb(OLD)
    );
  ELSIF TG_OP = 'UPDATE' THEN
    payload := jsonb_build_object(
      'type', 'UPDATE',
      'table', 'appointments',
      'record', to_jsonb(NEW),
      'old_record', to_jsonb(OLD)
    );
  ELSE
    payload := jsonb_build_object(
      'type', 'INSERT',
      'table', 'appointments',
      'record', to_jsonb(NEW)
    );
  END IF;

  PERFORM net.http_post(
    url := 'https://uhccgcojwqvkfyzrmebv.supabase.co/functions/v1/sync-gcal-webhook',
    body := payload,
    headers := '{"Content-Type": "application/json"}'::jsonb
  );

  RETURN NULL;
END;
$$;

-- 3. Trigger
DROP TRIGGER IF EXISTS sync_gcal_webhook ON appointments;
CREATE TRIGGER sync_gcal_webhook
  AFTER INSERT OR UPDATE OR DELETE ON appointments
  FOR EACH ROW
  EXECUTE FUNCTION sync_gcal_webhook_trigger();

CREATE OR REPLACE FUNCTION public.check_appointment_work_day_fn()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER

AS $$
DECLARE
    v_work_days    JSONB;
    v_dart_weekday INT;
    v_uid          UUID := auth.uid();

    -- Sem profissional atribuÃ­do: sem restriÃ§Ã£o de dias

        RETURN NEW;
    END IF;

            SELECT 1 FROM public.shops

            RETURN NEW;
        END IF;

    FROM public.employees e

    -- Profissional nÃ£o encontrado: deixa passar
    IF NOT FOUND THEN
        RETURN NEW;
    END IF;

    -- work_days NULL ou array vazio -> rejeitar

            'Configure os dias no painel do estabelecimento antes de aceitar agendamentos.';
    END IF;

    -- Converte DOW para padrÃ£o Dart (1=Seg â€¦ 6=SÃ¡b, 7=Dom)
    v_dart_weekday := CASE EXTRACT(DOW FROM NEW.scheduled_at)::INT
        WHEN 0 THEN 7   -- Domingo

    END;

    -- Valida se o dia estÃ¡ na lista de dias de trabalho
    IF NOT (v_work_days @> to_jsonb(v_dart_weekday)) THEN
        RAISE EXCEPTION
            'Agendamento bloqueado: o profissional nÃ£o trabalha neste dia da semana '
            '(dia %, work_days configurados: %).',

 LANGUAGE plpgsql

--      (fluxo "Completar Cadastro" apÃ³s Google OAuth)

-- 219_fix_shops_insert_rls.sql
-- A migration 218 reabilitou RLS na tabela shops, mas a polÃ­tica
-- "shops_owner_manage" (FOR ALL) da migration 193 foi perdida.
-- Restaram apenas:
--   - "shops_public_read" (SELECT pÃºblico) â€” OK
--   - "Shops are updatable by owners" (UPDATE apenas) â€” INSUFICIENTE
-- Sem INSERT, o upsert de criaÃ§Ã£o de loja no signup falha com RLS.
--
-- SoluÃ§Ã£o: recriar polÃ­tica FOR ALL que cobre INSERT/UPDATE/DELETE.

-- Garantir que helpers existem (criados em 193)
CREATE OR REPLACE FUNCTION public.is_admin()
RETURNS boolean LANGUAGE sql STABLE SECURITY DEFINER SET search_path = public
AS $$ SELECT EXISTS (SELECT 1 FROM public.users WHERE id = auth.uid() AND role = 'admin'); $$;

CREATE OR REPLACE FUNCTION public.is_shop_owner(target_shop_id UUID)
RETURNS boolean LANGUAGE sql STABLE SECURITY DEFINER SET search_path = public
AS $$ SELECT EXISTS (SELECT 1 FROM public.shops WHERE id = target_shop_id AND owner_id::text = auth.uid()::text); $$;

-- Recriar polÃ­tica completa (FOR ALL = SELECT + INSERT + UPDATE + DELETE)
DROP POLICY IF EXISTS "shops_owner_manage" ON public.shops;
DROP POLICY IF EXISTS "Shops are updatable by owners" ON public.shops;

CREATE POLICY "shops_owner_manage" ON public.shops
    FOR ALL TO authenticated
    USING (owner_id::text = auth.uid()::text OR public.is_admin())
    WITH CHECK (owner_id::text = auth.uid()::text OR public.is_admin());

CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  INSERT INTO public.users (
    id,
    name,
    email,
    phone,
    zip_code,
    role,
    created_at,
    updated_at
  )

      NEW.raw_user_meta_data->>'name',
      ''
    ),
    NEW.email,
    NEW.raw_user_meta_data->>'phone',
    NEW.raw_user_meta_data->>'zip_code',
    'cliente',
    now(),
    now()
  )
  ON CONFLICT (id) DO UPDATE SET
    -- Atualiza apenas dados nÃ£o sensÃ­veis de perfil

      WHEN EXCLUDED.name IS NOT NULL AND EXCLUDED.name != ''
      THEN EXCLUDED.name
      ELSE public.users.name
    END,
    email = COALESCE(EXCLUDED.email, public.users.email),
    updated_at = now()
    -- ROLE JAMAIS Ã‰ ATUALIZADO AQUI
    -- role permanece intacto para usuÃ¡rios existentes
  ;
